function loadDatabase() { let db = getDB(); let tbody = document.getElementById('database-table-body'); let searchQuery = document.getElementById('search-query').value.toLowerCase(); let filterOfficer = document.getElementById('filter-officer').value; let filterStatus = document.getElementById('filter-status').value; let today = new Date().toISOString().split('T')[0]; tbody.innerHTML = ''; // পারমিশন কন্ট্রোল: যদি ইউজার অ্যাডমিন না হয় (যেমন: অ্যাসিস্ট্যান্ট), তবে সে কেবল নিজের দায়িত্বপ্রাপ্ত কেস দেখবে let filtered = db.filter(item => { if(currentUser && currentUser.role !== 'admin') { // অ্যাসিস্ট্যান্ট কেবল তার নিজের অ্যাসাইন করা বা তার এলাকার কেস দেখবে if(item.assigned_officer !== currentUser.name && !item.assigned_officer.includes(currentUser.name)) { return false; } } let matchSearch = item.p_name.toLowerCase().includes(searchQuery) || item.case_no.toLowerCase().includes(searchQuery); let matchOfficer = true; if(filterOfficer === 'upazila_all') matchOfficer = item.assigned_officer !== 'জেলা প্রবেশন অফিসার, রাজশাহী'; else if(filterOfficer) matchOfficer = item.assigned_officer === filterOfficer; let status = item.end_date >= today ? 'Running' : 'Expired'; let matchStatus = !filterStatus || status === filterStatus; return matchSearch && matchOfficer && matchStatus; }); if(filtered.length === 0) { tbody.innerHTML = `কোনো রেকর্ড পাওয়া যায়নি বা আপনার এই ডেটা দেখার অনুমতি নেই।`; return; } filtered.forEach(item => { let isRunning = item.end_date >= today; let statusBadge = isRunning ? `চলমান` : `মেয়াদ উত্তীর্ণ`; // অ্যাডমিন ছাড়া সাধারণ ইউজার বা অ্যাসিস্ট্যান্টরা ডিলিট বা এডিট করতে পারবে কি না তার চেক let actionButtons = ` `; if(currentUser && currentUser.role === 'admin') { actionButtons += ``; } tbody.innerHTML += ` ${item.rcv_date} ${item.p_name}
পিতা: ${item.f_name} ${item.case_no}
${item.court_name} ${item.assigned_officer} ${item.start_date} হতে
${item.end_date} ${statusBadge} ${actionButtons} `; }); }
